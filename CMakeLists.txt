cmake_minimum_required(VERSION 3.16)
project(PPS CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

add_executable(PPS main.cpp)
target_include_directories(PPS PRIVATE include)
find_package(SFML 3 COMPONENTS Graphics QUIET)
if(SFML_FOUND)
  target_compile_definitions(PPS PRIVATE PPS_WITH_SFML)
  target_link_libraries(PPS PRIVATE SFML::Graphics)
  message(STATUS "SFML 3 found: building the viewer")
else()
  find_package(SFML 2.5 COMPONENTS graphics window system QUIET)
  if(SFML_FOUND)
    target_compile_definitions(PPS PRIVATE PPS_WITH_SFML)
    target_link_libraries(PPS PRIVATE sfml-graphics sfml-window sfml-system)
    message(STATUS "SFML 2 found: building the viewer")
  else()
    message(STATUS "SFML not found: building the headless runner only")
  endif()
endif()

enable_testing()
find_path(DOCTEST_INCLUDE doctest/doctest.h)
if(DOCTEST_INCLUDE)
  add_executable(test_pps tests/test_pps.cpp)
  target_include_directories(test_pps PRIVATE include ${DOCTEST_INCLUDE})
  add_test(NAME test_pps COMMAND test_pps)
endif()
